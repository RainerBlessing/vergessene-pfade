cmake_minimum_required(VERSION 3.20)
project(VergessenePfade VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(VP_BUILD_GAME "Build the SDL3 frontend" ON)
option(VP_VENDORED_SDL "Build SDL3 from vendor/ instead of using the system package" OFF)
set(VP_SDL_VENDOR_DIR ${CMAKE_SOURCE_DIR}/vendor/SDL3-3.2.28)

add_library(game_core src/world.c src/game.c src/content.c src/inventory.c src/combat.c)
target_include_directories(game_core PUBLIC src)

if(VP_BUILD_GAME)
    if(NOT VP_VENDORED_SDL)
        find_package(SDL3 CONFIG QUIET)
    endif()
    if(NOT SDL3_FOUND)
        if(NOT EXISTS ${VP_SDL_VENDOR_DIR}/CMakeLists.txt)
            message(FATAL_ERROR
                "SDL3 not found. Install the SDL3 development package, or unpack "
                "https://github.com/libsdl-org/SDL/releases/download/release-3.2.28/SDL3-3.2.28.tar.gz "
                "into vendor/ (see README).")
        endif()
        # Avoid a MinGW precompiled-header address-mapping failure.
        set(CMAKE_DISABLE_PRECOMPILE_HEADERS ON)
        set(SDL_SHARED OFF CACHE BOOL "" FORCE)
        set(SDL_STATIC ON CACHE BOOL "" FORCE)
        set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
        add_subdirectory(${VP_SDL_VENDOR_DIR} EXCLUDE_FROM_ALL)
    endif()
    add_executable(vergessene_pfade src/main.c src/renderer.c src/audio.c tests/journey.c)
    target_include_directories(vergessene_pfade PRIVATE tests)
    target_link_libraries(vergessene_pfade PRIVATE SDL3::SDL3 game_core)
    if(MINGW)
        target_link_options(vergessene_pfade PRIVATE -static-libgcc)
    endif()
    add_custom_target(copy_assets ALL
        COMMAND ${CMAKE_COMMAND} -E copy_directory
        ${CMAKE_SOURCE_DIR}/assets $<TARGET_FILE_DIR:vergessene_pfade>/assets)
    add_dependencies(vergessene_pfade copy_assets)
    add_executable(test_audio tests/test_audio.c src/audio.c)
    target_link_libraries(test_audio PRIVATE SDL3::SDL3 game_core)
    add_test(NAME sounds COMMAND test_audio ${CMAKE_SOURCE_DIR}/assets)
    install(TARGETS vergessene_pfade RUNTIME DESTINATION .)
    install(DIRECTORY assets DESTINATION .)
endif()

enable_testing()
add_executable(test_game tests/test_game.c)
target_link_libraries(test_game PRIVATE game_core)
add_test(NAME core COMMAND test_game ${CMAKE_SOURCE_DIR}/assets)
add_executable(test_journey tests/test_journey.c tests/journey.c)
target_link_libraries(test_journey PRIVATE game_core)
add_test(NAME complete_journey COMMAND test_journey ${CMAKE_SOURCE_DIR}/assets)
foreach(target game_core test_game test_journey test_audio vergessene_pfade)
    if(TARGET ${target})
        if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
            target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
        elseif(MSVC)
            target_compile_options(${target} PRIVATE /W4)
        endif()
    endif()
endforeach()
